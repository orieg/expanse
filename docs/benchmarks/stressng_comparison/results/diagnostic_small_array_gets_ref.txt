/* Diagnostic: gets on 500 JudyL arrays of about 20 keys, the shape of
 * stress-ng's sparsematrix hashjudy default cell (10,000 draws over 500 x 500).
 * mode "hit": every get replays a put's (row, col). mode "rand": fresh draws. */
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
typedef unsigned long Word_t;
extern void **JudyLIns(void **, Word_t, void *);
extern void **JudyLGet(const void *, Word_t, void *);
static uint64_t s = 0x9E3779B97F4A7C15ull;
static inline uint64_t rnd(void) { s ^= s << 13; s ^= s >> 7; s ^= s << 17; return s; }
#define ROWS 500
#define DRAWS 10000
int main(int argc, char **argv) {
    int hit = argc > 1 && !strcmp(argv[1], "hit");
    long reps = argc > 2 ? atol(argv[2]) : 2000;
    static void *arr[ROWS];
    static uint32_t px[DRAWS], py[DRAWS], gx[DRAWS], gy[DRAWS];
    for (int i = 0; i < DRAWS; i++) {
        px[i] = rnd() % ROWS; py[i] = rnd() % ROWS;
        void **v = JudyLIns(&arr[px[i]], py[i], 0);
        *(Word_t *)v = i + 1;
    }
    for (int i = 0; i < DRAWS; i++) {
        if (hit) { int j = rnd() % DRAWS; gx[i] = px[j]; gy[i] = py[j]; }
        else { gx[i] = rnd() % ROWS; gy[i] = rnd() % ROWS; }
    }
    uint64_t sum = 0, hits = 0;
    for (long r = 0; r < reps; r++)
        for (int i = 0; i < DRAWS; i++) {
            void **v = JudyLGet(arr[gx[i]], gy[i], 0);
            if (v) { sum += *(Word_t *)v; hits++; }
        }
    printf("%s reps=%ld gets=%ld hits=%lu sum=%lu\n", hit ? "hit" : "rand", reps, reps * DRAWS, hits, sum);
    return 0;
}
